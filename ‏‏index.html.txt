        function filterPersonnelTable() {
            const query = document.getElementById('search-input').value.toLowerCase();
            const filtered = personnelData.filter(p => 
                p.name.toLowerCase().includes(query) || 
                p.id.includes(query) || 
                p.code.toLowerCase().includes(query) ||
                p.rank.includes(query)
            );
            renderPersonnelTable(filtered);
        }

        // Bulk Import Parsing Algorithm
        function openBulkModal() { document.getElementById('bulk-modal').classList.remove('hidden'); }
        function closeBulkModal() { document.getElementById('bulk-modal').classList.add('hidden'); }

        function processBulkImport() {
            const text = document.getElementById('bulk-text-input').value.trim();
            if (!text) return;

            // Regex pattern to extract discord tags, IDs, Codes, and Names
            const tagMatch = text.match(/<@\d+>/g) || [];
            const idMatch = text.match(/\d{5,7}/g) || [];
            const codeMatch = text.match(/R\s*-\s*\d+/gi) || [];

            if (tagMatch.length > 0 && idMatch.length > 0) {
                for (let i = 0; i < tagMatch.length; i++) {
                    const discordTag = tagMatch[i] || "";
                    const id = idMatch[i] || Math.floor(100000 + Math.random() * 900000).toString();
                    const code = codeMatch[i] ? codeMatch[i].replace(/\s+/g, '') : "R-100";

                    // Check if exists
                    if (!personnelData.some(p => p.id === id)) {
                        personnelData.push({
                            id: id,
                            name: `منسوب_${id}`,
                            rank: "وكيل رقيب",
                            discordTag: discordTag,
                            code: code,
                            records: 0,
                            announcements: 0
                        });
                    }
                }
                renderPersonnelTable();
                showToast('تم إدراج وتحليل المنسوبين الجدد بنجاح!');
                closeBulkModal();
            } else {
                showToast('تعذر تحليل النص! تأكد من الصيغة.', 'error');
            }
        }

        // Audit Processing (25,2,4 or Full Text)
        function processAuditData() {
            const text = document.getElementById('audit-text-input').value.trim();
            if (!text) return;

            const lines = text.split('\n');
            let updatedCount = 0;

            lines.forEach(line => {
                // Short Format: 25,2,4
                const shortMatch = line.match(/^(\d+),(\d+),(\d+)$/);
                if (shortMatch) {
                    const [, targetId, recs, anns] = shortMatch;
                    const person = personnelData.find(p => p.id === targetId);
                    if (person) {
                        person.records += parseInt(recs);
                        person.announcements += parseInt(anns);
                        updatedCount++;
                    }
                }
            });

            if (updatedCount > 0) {
                renderPersonnelTable();
                showToast(`تم تحديث سجلات ${updatedCount} منسوب بنجاح!`);
                document.getElementById('audit-text-input').value = '';
            } else {
                showToast('لم يتم العثور على أيديات مطابقة!', 'error');
            }
        }

        // Shift Audit Processing
        function processShiftAudit() {
            const text = document.getElementById('shift-text-input').value.trim();
            if (!text) return;

            const ids = text.match(/\d{3,6}/g) || [];
            let countMap = {};

            ids.forEach(id => { countMap[id] = (countMap[id] || 0) + 1; });

            Object.keys(countMap).forEach(targetId => {
                const person = personnelData.find(p => p.id === targetId);
                if (person) {
                    person.records += countMap[targetId];
                }
            });

            renderPersonnelTable();
            showToast('تم احتساب تكرار الفترات وتحديث السجلات بنجاح!');
            document.getElementById('shift-text-input').value = '';
        }

        // Promotions & Auto-Code Allocation
        function renderPromotions() {
            const container = document.getElementById('promotion-cards-container');
            container.innerHTML = '';

            personnelData.forEach(person => {
                const card = document.createElement('div');
                card.className = "bg-dark-800 p-4 rounded-xl border border-white/5 space-y-2";
                card.innerHTML = `
                    <div class="flex items-center justify-between">
                        <span class="font-bold text-white text-xs">${person.name} (${person.id})</span>
                        <span class="text-[10px] text-grave-gold bg-grave-gold/10 px-2 py-0.5 rounded">${person.rank}</span>
                    </div>
                    <p class="text-[11px] text-gray-400 font-mono">${person.discordTag}</p>
                    <div class="flex items-center justify-between pt-2 border-t border-white/5 text-[11px]">
                        <span class="text-gray-400">السجلات: <strong class="text-white">${person.records}</strong></span>
                        <span class="text-emerald-400 font-bold">مستحق ترقية</span>
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function copyEligiblePromotions() {
            let copyText = "مستحقي الترقيات:\n\n";
            personnelData.forEach(p => {
                copyText += `${p.discordTag}\nGR💂🏻‍♀️${p.code} ${p.name} ${p.id}\n---\n`;
            });

            navigator.clipboard.writeText(copyText);
            showToast('تم نسخ لستة المستحقين حافظة الجهاز!');
        }
